Arithmetic Operators

var a=5
var b=4

var sum = a + b
print("Addition:",sum)

var sub=a-b
print("Subtraction:",sub)

var mul=a*b
print("Multiply:",mul)

var div=a/b
print("Divide:",div)

var mod=a%b
print("Modulus:",mod)


Output:

Addition: 9
Subtraction: 1
Multiply: 20
Divide: 1
Modulus: 1

Range Operators

for i in 1...3 {
    print("closed = \(i)")
}

for i in 1..<3 {
    print("half = \(i)")
}

let values = [ 1, 2, 3, 4 ]
for i in values[0...] 
{
    print("One sided = \(i)")
}

Output:

closed = 1
closed = 2
closed = 3
half = 1
half = 2
One sided = 1
One sided = 2
One sided = 3
One sided = 4
